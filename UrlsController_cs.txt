using Microsoft.AspNetCore.Mvc;
using UrlShortener.Contracts;
using UrlShortener.Services;

namespace UrlShortener.Controllers;

[ApiController]
[Route("api/urls")]
public sealed class UrlsController(IUrlShortenerService service) : ControllerBase
{
    [HttpPost]
    public async Task<ActionResult<CreateShortUrlResponse>> Create([FromBody] CreateShortUrlRequest request)
    {
        var baseUrl = $"{Request.Scheme}://{Request.Host}";
        var result = await service.CreateAsync(request, baseUrl);
        return Ok(result);
    }

    [HttpGet("{code}/analytics")]
    public async Task<ActionResult<UrlAnalyticsResponse>> GetAnalytics(string code)
    {
        var result = await service.GetAnalyticsAsync(code);
        return result is null ? NotFound() : Ok(result);
    }
}
