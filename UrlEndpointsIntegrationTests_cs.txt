using NUnit.Framework;
using System.Net;
using System.Net.Http.Json;
using System.Text.Json;
using Microsoft.AspNetCore.Mvc.Testing;

namespace UrlShortener.Tests.Integration;

[TestFixture]
public sealed class UrlEndpointsIntegrationTests
{
    private CustomWebApplicationFactory _factory = null!;
    private HttpClient _client = null!;

    [SetUp]
    public void SetUp()
    {
        _factory = new CustomWebApplicationFactory();
        _client = _factory.CreateClient(new WebApplicationFactoryClientOptions
        {
            AllowAutoRedirect = false
        });
    }

    [TearDown]
    public void TearDown()
    {
        _client.Dispose();
        _factory.Dispose();
    }

    [Test]
    public async Task CreateUrl_ThenRedirect_ThenAnalytics_ShouldTrackClickCount()
    {
        var createResponse = await _client.PostAsJsonAsync("/api/urls", new
        {
            url = "https://example.com/docs"
        });

        Assert.That(createResponse.StatusCode, Is.EqualTo(HttpStatusCode.OK));

        var createPayload = await createResponse.Content.ReadAsStringAsync();
        using var createJson = JsonDocument.Parse(createPayload);

        var code = createJson.RootElement.GetProperty("code").GetString();
        Assert.That(code, Is.Not.Null.And.Not.Empty);

        var redirectResponse = await _client.GetAsync($"/{code}");
        Assert.That(redirectResponse.StatusCode, Is.EqualTo(HttpStatusCode.Redirect));
        Assert.That(redirectResponse.Headers.Location, Is.Not.Null);
        Assert.That(redirectResponse.Headers.Location!.ToString(), Is.EqualTo("https://example.com/docs"));

        var analyticsResponse = await _client.GetAsync($"/api/urls/{code}/analytics");
        Assert.That(analyticsResponse.StatusCode, Is.EqualTo(HttpStatusCode.OK));

        var analyticsPayload = await analyticsResponse.Content.ReadAsStringAsync();
        using var analyticsJson = JsonDocument.Parse(analyticsPayload);

        var clickCount = analyticsJson.RootElement.GetProperty("clickCount").GetInt32();
        Assert.That(clickCount, Is.EqualTo(1));
    }

    [Test]
    public async Task UnknownCodeRedirect_ShouldReturnNotFound()
    {
        var response = await _client.GetAsync("/does-not-exist");
        Assert.That(response.StatusCode, Is.EqualTo(HttpStatusCode.NotFound));
    }
}
